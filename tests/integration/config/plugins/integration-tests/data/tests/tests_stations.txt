test "Industry - Found A Station"
	status active
	description "Tests founding a space station from New Greenland, then landing on it and building on it."
	sequence
		inject "New Greenland Industry Save"
		call "Load First Savegame"
		# Get enough credits for a station.
		input
			key "F12"
		input
			key 5
		input
			key 5
		input
			key 5
		input
			key "escape"
		# Stations are listed first. Found one, keeping the suggested name.
		input
			key n
		input
			key e
		input
			key "Return"
		assert
			and
				"facility: Orbital Station" == 1
				"credits" == 631000
		# Fly to the new station and land on it.
		call "Depart"
		navigate
			"travel destination" "Arcturus Station"
		call "Land"
		assert
			"flagship planet: Arcturus Station" > 0
		# Store the fleet's cargo in the station's warehouse. The warehouse is the
		# first row of the list.
		input
			key n
		input
			key u
		assert
			and
				"commodity: Plastic" == 0
				"facilities owned" == 1
