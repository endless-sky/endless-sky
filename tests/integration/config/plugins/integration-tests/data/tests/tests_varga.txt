test "Varga Claims - Core Samples Delivery"
	status active
	description "Tests that the core samples quest takes the produced goods and gives its exclusive rewards."
	sequence
		inject "Varga Claims Save"
		call "Load First Savegame"
		assert
			and
				"commodity: Core Samples" == 25
				"outfit: Varga Ore Hopper" == 0
		# Landing missions are created when landing, so take off and land again.
		# Edda offers the quest; accept it.
		call "Depart"
		navigate
			"travel destination" "New Greenland"
		call "Land"
		input
			key "Return"
		assert
			"Varga Claims 2: offered" > 0
		# Land again with the samples to complete the quest.
		call "Depart"
		navigate
			"travel destination" "New Greenland"
		call "Land"
		assert
			and
				"Varga Claims 2: done" > 0
				"commodity: Core Samples" == 5
				"outfit: Varga Ore Hopper" == 1
				"expanded: crystal bore" > 0
