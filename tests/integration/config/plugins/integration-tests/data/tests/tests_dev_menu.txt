test "Dev Menu - Credits And Days"
	status active
	description "Tests that the developer menu on the planet screen adds credits and advances the date."
	sequence
		inject "Three Earthly Barges Save"
		call "Load First Savegame"
		assert
			and
				"flagship landed" > 0
				"credits" == 131000
				"day" == 16
		# Open the developer menu and add 100,000 credits.
		input
			key "F12"
		input
			key 4
		assert
			"credits" == 231000
		# Advance one day. (Salaries are paid, so only check the date.)
		input
			key 1
		assert
			"day" == 17
		# Close the menu again; the player should still be on the planet.
		input
			key "escape"
		assert
			"flagship landed" > 0


test "Dev Menu - In Flight"
	status active
	description "Tests that the developer menu can also be opened in flight."
	sequence
		inject "Three Earthly Barges Save"
		call "Load First Savegame"
		call "Depart"
		apply
			"test: credits before" = "credits"
		input
			key "F12"
		input
			key 4
		input
			key "escape"
		assert
			and
				"flagship landed" == 0
				"credits" == "test: credits before" + 100000
