test "Industry - Build And Collect"
	status active
	description "Tests building a Mining Outpost on New Greenland, letting it produce for a week, and collecting the goods."
	sequence
		inject "New Greenland Industry Save"
		call "Load First Savegame"
		assert
			and
				"flagship planet: New Greenland" > 0
				"credits" == 131000
				"facilities owned" == 0
		# The list has stations first, then the rest alphabetically: Orbital Station,
		# Hydroponics Farm, Machine Works, Mining Outpost, Textile Mill.
		# Select the Mining Outpost. There are not enough credits yet, so building does nothing.
		input
			key n
		input
			key down
		input
			key down
		input
			key down
		input
			key e
		assert
			and
				"credits" == 131000
				"facilities owned" == 0
		# Add 100,000 credits with the developer menu, then build.
		input
			key "F12"
		input
			key 4
		input
			key "escape"
		input
			key e
		assert
			and
				"credits" == 81000
				"facilities owned" == 1
		# Let a week pass (4 tons a day), then collect the 28 tons of Metal.
		input
			key "F12"
		input
			key 2
		input
			key "escape"
		input
			key c
		# Build a Machine Works and supply it with the Metal and the 10 tons of
		# Plastic the fleet started with.
		input
			key "F12"
		input
			key 5
		input
			key "escape"
		input
			key up
		input
			key e
		input
			key u
		assert
			"facilities owned" == 2
		# After one day the Machine Works has turned 2 Metal and 1 Plastic into 4 Equipment.
		input
			key "F12"
		input
			key 1
		input
			key "escape"
		input
			key c
		# Take off and check that only the 4 tons of Equipment are aboard: the
		# Metal and Plastic went into the Machine Works.
		apply
			"test: cargo space before" = "cargo space free"
		call "Depart"
		assert
			and
				"flagship landed" == 0
				"cargo space free" == "test: cargo space before" - 4
				"facilities owned" == 2
